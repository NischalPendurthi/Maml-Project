\section{Problem}


%====================================================================
\begin{frame}{Problem statement}
\footnotesize
\begin{block}{Setting}
$N$ agents on a graph $\mathcal G$, \textbf{no server}. Each round agent $i$ sees $K$ contexts $x\in\mathbb R^d$, pulls one, observes $y=f(\langle x,\theta^*\rangle)+\eta$. Unknown, shared $\theta^*$ and $f$ ($f$ possibly non-monotone). Goal: small network regret $R_T(N)$, little communication.
\end{block}
\begin{block}{Difficulties}
\begin{itemize}
\item \textbf{Two unknowns:} the direction needs pooled data; the shape of $f$ needs per-bin counts.
\item \textbf{No coordinator:} totals are built over the graph, $\approx$ diameter rounds late.
\item \textbf{Stale or averaged counts} break the UCB confidence argument.
\item \textbf{Communication:} sharing tables every round costs $\Theta(|E|N_bT)$ scalars.
\end{itemize}
\end{block}
\begin{alertblock}{What prior work misses}
Single-index bandits are single-agent; decentralised bandits assume finite arms, linear or kernel models. No decentralised single-index bandit, and no graph-dependent lower bound for this class.
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Contributions $\star$}
\small
\begin{enumerate}
\setlength{\itemsep}{0.35em}
\item \textbf{New problem:} the first decentralised single-index bandit. Agents share $\theta^*$ and $f$ and talk only over a graph.
\item \textbf{Lower bounds:} no serverless algorithm beats $\Omega((NT)^{2/3})$ (Thm 5). On rings and paths it must also pay $\Omega(N\min(T,D)^{1/3})$ (Thm 7, light cones).
\item \textbf{Upper bound for the algorithm we run:} the published-only relay has regret $\tilde O\big((dNT)^{2/3}+KF'\big)$ (Thm 27): the server's rate plus an \emph{additive, horizon-free} graph term set by the farness $F'$. It holds for every trigger threshold $\gamma$, so $\gamma$ trades regret against communication.
\item \textbf{Communication and robustness:} logarithmic in $T$ for a fixed grid (Thm 29), $O(N_b+N)$ memory per agent. Directed, delayed and time-varying graphs change only the delivery time, and our bound $B(N-1)$ is tight.
\item \textbf{Experiments:} 37 studies with paired trials and 95\% CIs, run in the order the problems came up. The measured graph cost has the proved shape, bin owners remove what is left of it, and we compare with prior papers' algorithms on our problem.
\end{enumerate}
\end{frame}

%====================================================================
\begin{frame}{Recap: single-index bandits}
\small
\begin{block}{Model}
Each round an agent sees $K$ arms $x_{t,a}\in\mathbb{R}^d$, pulls one, and observes
\[
\boxed{y_t=f(\langle x_t,\theta_*\rangle)+\eta_t}\qquad\theta_*\text{ and }f\text{ unknown},\ f\text{ possibly non-monotone.}
\]
\end{block}
\begin{block}{ZoomSIB-UCB (Dey, Bhore \& Ghosh 2026): optimal $\tilde\Theta(T^{2/3})$}
\begin{itemize}
\item \textbf{Phase 1:} explore; estimate $\theta_*$ with the Stein estimator $\hat v=\frac1{T_0}\sum_t y_tS(x_t)\propto\theta_*$, a \emph{sample mean}.
\item \textbf{Phase 2:} project $z=x^\top\hat\theta$, cut the line into bins, run \emph{sleeping} UCB over the bins available each round. State: $(n_j,S_j)$ per bin.
\end{itemize}
\end{block}
\begin{alertblock}{Our question}
\centering $N$ agents share $\theta_*$ and $f$ but \textbf{no server}: they talk only to graph neighbours.
\end{alertblock}
\end{frame}

No trusted coordinator, or no backbone: sensor networks, robot/drone swarms, vehicular networks, peer-to-peer recommendation, consortia that do not share raw data.
\end{block}

\end{center}

\begin{frame}{Experimental setup}
\footnotesize
\begin{block}{Instances (as in Dey, Bhore \& Ghosh)}
$d=10$, $K=20$ contexts $\sim\mathcal N(0,I)$, noise $\sigma=0.1$. Links: quadratic $-(z-1)^2+1$ (non-monotone), zigzag $\sin z+0.3z$, logistic (monotone).
\end{block}
\begin{block}{Networks}
$N=4$--$32$ agents (16 by default) on complete, expander, hypercube, torus, ring, path graphs ($D=1$--$15$), plus failing, delayed, time-varying and directed links.
\end{block}
\begin{block}{Protocol}
\begin{itemize}
\item $T\le20{,}000$ per agent ($NT\le320$k); pooled Phase 1 $=960\,(NT/160\text{k})^{2/3}$; $\Delta=(NT)^{-1/3}$; $\gamma=0.5$ unless stated.
\item 10--20 \textbf{paired} trials (the same seeds for every method), mean $\pm$ 95\% CI.
\item Metrics: network regret, \emph{excess} over the server, scalars sent.
\item References: server, independent agents. Baselines: consensus / gossip / push-sum protocols; prior algorithms with ideal relaying.
\end{itemize}
\end{block}
\end{frame}

%====================================================================
\begin{frame}{Step 1: reproduce ZoomSIB with one agent (E1--E5)}
\footnotesize
\begin{block}{What we checked}
\begin{itemize}
\item Stein direction error falls like $n^{-1/2}$: slope $-0.50$ for $d=5$--$40$ (E1).
\item ZoomSIB beats LinUCB, ESTOR and GSTOR on all three non-monotone links. ESTOR is even worse than LinUCB on the quadratic link: the price of assuming $f$ is monotone (E2).
\item Log-log regret slope $0.29$--$0.60$, below the theory's $2/3$ (E3).
\item Phase 1 needs only $\approx2\%$ of the horizon; adaptive stopping exits at $1.7\%$ (E5).
\end{itemize}
\end{block}
\begin{alertblock}{What we took from it}
The learning state is a direction estimate and a per-bin table $(n_j,S_j)$, and both are sums. Agents can add sums exactly, so we decided to share sums. Our pitch had planned to average each agent's fitted estimate instead.
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Step 3: remove the server: mix or relay?}
\scriptsize
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Phase 1: build $\sum_kb_k$ over the graph}
\begin{itemize}
\item DGD, decentralised FedAvg: stay \textbf{biased} by $4$--$46\%$; on the star, regret $103{,}555$ vs the server's $9{,}674$.
\item Gradient tracking: unbiased, $208$k scalars.
\item \textbf{Spanning tree:} exact, $2.6$k scalars.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Phase 2: share the bin tables}
\begin{itemize}
\item Consensus, push-sum: server's regret, but $140$--$285$M scalars.
\item Burst gossip: $+4$--$13\%$; effective-sample-size counts no better than naive ones.
\item One-hop neighbours only: $+20$--$91\%$.
\item \textbf{Event-triggered flooding:} $+2$--$5\%$, $70$k scalars.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.4em}
\begin{alertblock}{What we learned}
Averaging (mixing) only converges towards the sum. Relaying exact, sparse records delivers the sum itself, a few rounds late, and gives the same regret with about $2000\times$ fewer scalars. Every later design relays.
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Step 3, continued: a misleading win}
\footnotesize
\begin{block}{Surprise: the network beat the server}
With adaptive Phase-1 stopping, push-sum on a torus had regret $5{,}761$, \emph{below} the server's $9{,}719$ (E16). Agents that disagree keep exploring longer, so the gain came from the stopping rule and not from the network.
\end{block}
\begin{block}{Fix: compare at a fixed $T_0$ (every later experiment)}
Then relays track the diameter ($+1\%$ at $D{=}1$, $+15\%$ at $D{=}15$) and mixing tracks the spectral gap.
\end{block}
\begin{block}{Second surprise: acceleration is fragile (E18)}
Chebyshev gossip: $\times2$ at $30\%$ link failures, $\times7$ at $90\%$. Flooding: $+27\%$ at $90\%$, because it only needs every message to arrive eventually.
\end{block}
\begin{alertblock}{Design after Step 3}
Spanning-tree Phase 1 + event-triggered flood relay: $\times1.030$ of the server with $60$k scalars, best trade-off of 28 designs (E20).
\end{alertblock}
\end{frame}

\section{Theory}


%====================================================================
\begin{frame}{Theory: setting of the proofs}
\footnotesize
\begin{block}{Assumptions (DBG's, with one made explicit)}
\begin{itemize}
\item Contexts i.i.d., coordinates $\kappa$-sub-Gaussian, $\mathbb E[S_j(X)^2]\le M$; $|f|\le L_f$, $|f'|\le L_{f'}$ on $|z|\le X_{\max}$; $\mu^*=\mathbb E f'(\langle X,\theta^*\rangle)>0$.
\item Noise i.i.d.\ and $\sigma$-sub-Gaussian. DBG's Stein lemma already needs i.i.d.\ samples, and their hard instance has $\mathcal N(0,1)$ noise.
\end{itemize}
\end{block}
\begin{block}{Communication: messages move one hop per round}
A flooded message from agent $l$ reaches agent $i$ after $d(i,l)$ rounds, and everyone within the \textbf{delivery time} $\Lambda$ ($=D$, the diameter, on a static graph). The regret bound uses the distances through the \textbf{farness} $F'=\max_i\sum_l\max(d(i,l),1)\le ND$; robustness results replace them by $\Lambda$.
\end{block}
\begin{block}{Order of the theory (\texttt{docs/proofs/serverless.pdf})}
No algorithm beats (Thm 5, 7) $\to$ ours achieves (Thm 27) $\to$ it sends (Thm 29) $\to$ bad networks (Lemma 31) $\to$ agents must agree (Thm 33).
\end{block}
\end{frame}

%====================================================================
\begin{frame}{Theorem 5: no serverless algorithm beats $(NT)^{2/3}$}
\footnotesize
\begin{block}{Theorem}
For any $N$-agent algorithm, on any graph, even one where every agent sees all data after every round:\quad
$\displaystyle\sup_{\text{instances}}\mathbb E\,R_T(N)\ \ge\ c\,(NT)^{2/3}.$
\end{block}
\textbf{Proof idea: one agent can simulate the whole network.}
\begin{center}
\begin{tikzpicture}[x=0.5cm,y=0.5cm,font=\scriptsize]
\foreach \t/\r in {0/1,1/2,2/3}{
  \foreach \i in {1,...,4}{
    \node[draw,circle,minimum size=0.45cm,inner sep=0pt,fill=blue!10] at ({\t*5+\i},0) {\i};
  }
  \node at ({\t*5+2.5},1) {round \r};
  \draw[->,thick,orange] ({\t*5+4.4},-0.4) -- ({\t*5+5.6},-0.4) node[midway,below] {rewards};
}
\node[anchor=west] at (15.3,0) {$\cdots$};
\end{tikzpicture}
\end{center}
\begin{itemize}
\item Single-agent round $u=(t-1)N+i$ plays what agent $i$ would play in round $t$.
\item It \textbf{withholds} round $t$'s rewards until all $N$ agents of round $t$ have acted, then runs one communication step.
\item Same law of the history $\Rightarrow$ network regret $=$ regret of one agent over $NT$ rounds $\ge c(NT)^{2/3}$ (DBG Thm 5.1).
\end{itemize}
\vspace{-0.3em}{\tiny Hard instances: DBG's at horizon $NT$ ($d=1$, Gaussian contexts, $K\approx(NT)^{1/3}\log NT$ arms).}
\end{frame}

%====================================================================
\begin{frame}{Theorem 7: the graph does cost regret (light cones)}
\scriptsize
\begin{columns}[c]
\begin{column}{0.46\textwidth}
\centering
\begin{tikzpicture}[x=0.24cm,y=0.36cm,font=\tiny]
\draw[->] (0,0) -- (19,0) node[right] {agents};
\draw[->] (0,0) -- (0,5.6) node[above] {round};
\foreach \c in {3,9.5,16}{
  \fill[blue!20] (\c-3,0.2) -- (\c+3,0.2) -- (\c,5) -- cycle;
  \draw[blue!60!black] (\c-3,0.2) -- (\c,5) -- (\c+3,0.2);
  \fill (\c,0.2) circle (1.5pt);
}
\node[align=center] at (9.5,-1.2) {disjoint cones of radius $R$:\\ $\approx R^2$ agent-rounds each};
\end{tikzpicture}
\end{column}
\begin{column}{0.52\textwidth}
\begin{block}{Lemma 6 (light cones are closed)}
Information moves one hop per round, so the actions inside a cone depend only on data inside it.
\end{block}
\end{column}
\end{columns}
\begin{block}{Theorem 7 and Corollary 8}
Each cone acts like one agent with $n\approx R^2$ pulls, so it pays $\Omega(n^{2/3})$. One common hard instance makes all the cones pay at once. On a ring or a path:
\[
\sup\mathbb E\,R_T(N)\ \ge\ c\,\max\Big\{(NT)^{2/3},\ \ N\min(T,D)^{1/3}\Big\}.
\]
\end{block}
\begin{itemize}
\item The second term wins when $T\lesssim\sqrt{ND}$: there, \textbf{no network on a ring can match one agent with $NT$ pulls}.
\item Longer horizons: upper graph term $KF'$ (Thm 27) vs this $ND^{1/3}$. Linear in distance necessary? \alert{Open}: a cone already learns at rate $n^{2/3}$ on its own data.
\end{itemize}
\end{frame}

%====================================================================
\begin{frame}{Step 5: the first proof breaks; epochs fix it}
\scriptsize
\begin{block}{The gap in the first proof}
In the flood relay, an agent uses its own unpublished rewards. A reward changes the agent's next pulls, which change \emph{when} its trigger fires, i.e.\ when the pull is published. Each agent's view became a union of $N$ per-agent prefixes, and the proof lost a factor $\sqrt N$ in the confidence width, which breaks the matching rate.
\end{block}
\begin{block}{First fix: synchronised epochs (Theorems 17, 22; backup)}
Every agent acts on the \emph{same} delayed table plus its own pulls: one common prefix plus one own block, so the server's width. Proved: $\tilde O\big((dNT)^{2/3}+N\Lambda K\big)$, horizon-free. The measured excess divided by $NDK$ is $0.86$--$1.15$, so the bound has the right shape.
\end{block}
\begin{alertblock}{But it is slow (E31)}
$+5\%$ to $+59\%$ over the server, against $+1.5\%$ to $+9\%$ for the relay we ran: $3$--$6\times$ the extra regret. Every agent waits for the farthest agent ($\approx4D$ rounds).
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Step 6: the algorithm we now run}
\scriptsize
\begin{block}{Every agent $i$ runs the same code}
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{@{}r@{\hspace{0.6em}}l@{}}
\multicolumn{2}{@{}l}{\textbf{Phase 1: learn the direction}}\\
1: & \textbf{for} $t\le T_0$: pull a random arm; $b_i\leftarrow b_i+\varphi_\tau\big(y\,S(x)\big)$ \quad$\triangleright$ truncated Stein sum\\
2: & flood $b_i$; wait $\Lambda$ rounds \quad$\triangleright$ every agent now holds all $N$ sums\\
3: & $\hat\theta\leftarrow\sum_kb_k/\|\sum_kb_k\|_1$ \quad$\triangleright$ identical everywhere; equals the server's\\
\multicolumn{2}{@{}l}{\textbf{Phase 2: learn $f$ with sleeping UCB on bins of width $\Delta$ along $\langle x,\hat\theta\rangle$}}\\
4: & \textbf{for} each round $t$:\\
5: & \quad $(n_j,S_j)\leftarrow\sum_k$ freshest \emph{published} table of $k$ \quad$\triangleright$ own table too, once published\\
6: & \quad pull an arm in the available bin with the largest $S_j/n_j+c_\delta/\sqrt{n_j}$\\
7: & \quad \textbf{if} own unpublished pulls in some bin $j$ exceed $\gamma\max(n_j,1)$: flood own table\\
\end{tabular}
\end{block}
\begin{alertblock}{Why this design}
\begin{itemize}
\item Phase 1 is \textbf{exact}: Stein sums are sufficient statistics, so flooding them gives the server's $\hat\theta$.
\item Phase 2 relays \textbf{exact, sparse} tables, published only when a bin grows by a factor $\gamma$.
\item Acting only on \emph{published} data is what makes Theorem 27 provable.
\end{itemize}
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Step 6: Theorem 27, regret of the algorithm we run}
\scriptsize
\begin{block}{Theorem 27 (published-only relay, any $\gamma>0$, w.p.\ $\ge1-4\delta$)}
\vspace{-0.6em}
\begin{align*}
R_T(N)\le{}&\underbrace{2L_fN(T_0{+}\Lambda)}_{\text{Phase 1}}
+\underbrace{4\sqrt2\,c'_\delta\sqrt{(1{+}N\gamma)N_bNT}}_{\text{UCB}}
+\underbrace{2L_{f'}\Delta NT}_{\text{bin width}}
+\underbrace{4L_{f'}X_{\max}\varepsilon_0 NT}_{\text{direction error}}\\[-0.2em]
&+\underbrace{2L_f\big(4F'\min(K,N_b)+N_b(6\ell'+N\gamma+1)\big)}_{\text{delay: the only graph term}}
\end{align*}
\end{block}
\begin{block}{Rates ($\Delta=(NT)^{-1/3}$, $T_0$ balanced)}
\begin{itemize}
\item $\gamma=\gamma_0/N$: $R_T(N)=\tilde O\big((dNT)^{2/3}+K F'\big)$; matches Thm 5 when $KF'\le(NT)^{2/3}$.
\item constant $\gamma$ (we run $0.5$): $\tilde O\big((dNT)^{2/3}+\sqrt{1+N\gamma}\,(NT)^{2/3}+KF'\big)$, $N\times$ fewer publications.
\end{itemize}
\end{block}
\begin{alertblock}{Reading the bound}
The graph adds a term that does not grow with $T$ and is set by the sum of distances: star $F'\approx2N$, ring $F'\approx ND/2$, path $F'\le ND$.
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Why Theorem 27 holds: three ideas}
\footnotesize
\begin{enumerate}
\setlength{\itemsep}{0.4em}
\item \textbf{One prefix, no $\sqrt N$ (Lemmas 24, 25).} Agents act only on \emph{published} records, their own included. So a reward cannot change \emph{when} its pull is published, and anything it influences reaches every agent later (triangle inequality). Each agent's view is one prefix of a fixed order: a martingale, so the confidence width is the server's.
\item \textbf{Views nest (Lemma 26).} What agent $l$ saw by round $s$, agent $i$ sees by round $s+d(i,l)$. So $l$'s unpublished pulls are at most $\gamma\times$ a count that $i$ \emph{also} sees, and after a short lag $i$ sees a $1/(2(1{+}N\gamma))$ fraction of the true count, whatever $\gamma$ is.
\item \textbf{Only $K$ arms (Lemma 21).} A pull into bin $j$ needs an arm in bin $j$. In the $\le F'$ agent-rounds that $i$ cannot yet see, only $K$ arms per agent-round exist, so in total $\approx KF'$ pulls are unseen, not $F'$ per bin, and the term does not grow with $T$.
\end{enumerate}
\vspace{0.3em}
\begin{alertblock}{Step 6, continued: from $\gamma=1/(2N)$ to any $\gamma$}
Our first version of this proof needed $N\gamma\le1/2$, i.e.\ $\gamma=1/(2N)$: $7$--$8\times$ the scalars of the relay we ran (E36). Idea 2 removed the restriction, so $\gamma=0.5$, the setting we run, is now proved.
\end{alertblock}
\end{frame}

\begin{frame}{Step 7: bin owners remove the rest of the gap (E37)}
\scriptsize
\centering\includegraphics[width=0.8\textwidth]{e37_owners.png}

\begin{itemize}
\setlength{\itemsep}{1pt}
\item \textbf{Where the $KF'$ term comes from:} far-apart agents explore the same \emph{empty} bin before they hear of each other. Such blind pulls grow with the farness (server $412$, ring $543$, path $581$).

\begin{frame}{Theorem 29: communication}
\footnotesize
\begin{block}{Theorem (published-only relay; the same count holds for epochs)}
Each flooded message crosses each edge at most once, and each agent publishes at most $N_b\big(1+\log_{1+\gamma}(NT)\big)$ times, so the number of scalars is at most
\[
\underbrace{2|E|N(d{+}1)}_{\text{Phase 1}}+\underbrace{2|E|\,N\,N_b\big(1+\log_{1+\gamma}(NT)\big)(3N_b{+}1)}_{\text{Phase 2}} .
\]
\end{block}
\begin{itemize}
\item \textbf{Why logarithmic:} between two publications triggered by a bin, its count grows by a factor $>1+\gamma$.
\item For a fixed grid this is \textbf{logarithmic in $T$}, against $\Theta(|E|N_bT)$ for per-round sharing.
\end{itemize}
\begin{alertblock}{Choosing $\gamma$ (Thm 27)}
$\gamma=\gamma_0/N$ gives the optimal rate but $\approx N\times$ more publications ($\tilde O(|E|N^2(NT)^{2/3})$ at the rate-optimal grid). A constant $\gamma$ publishes $N\times$ less and the bound pays $\sqrt{1+N\gamma}$. Measured (E36, ring): $\gamma=0.5$ sends $180$k scalars for $+2\%$ regret; $\gamma=1/32$ sends $1.37$M.
\end{alertblock}
\end{frame}

\begin{frame}{Theorem 33: agents must agree on the direction}
\footnotesize
\begin{block}{Theorem}
If agent $k$ bins with its own $\hat\theta_k$ and $\max_k\|\hat\theta_k-\hat\theta\|_1\le\eta_{\rm agree}$ ($\hat\theta$ = exact pooled), the regret grows by at most
\[
4L_{f'}X_{\max}\,\eta_{\rm agree}\,NT .
\]
\end{block}
\begin{itemize}
\item \textbf{Why:} directions enter the proof only through the projection error. Pooling a bin across agents still pools pulls whose means lie in the same interval.
\item \textbf{Linear in $NT$:} even small disagreement hurts. It matters once $X_{\max}\eta_{\rm agree}NT\sim(NT)^{2/3}$.
\item \textbf{Fix:} one extra flood of the local Stein sums gives every agent the exact direction ($\eta_{\rm agree}=0$), for $\Lambda$ rounds and $2|E|N(d{+}1)$ scalars.
\end{itemize}
\begin{block}{Measured}
Consensus Phase 1: $\times1.25$ (ring), $\times1.34$ (path). One flooding pass: $\times1.08$, $\times1.09$.
\end{block}
\end{frame}

\begin{frame}{Conclusion and future work}
\scriptsize
\setlength{\itemsep}{0pt}
\begin{block}{Conclusion}
\begin{itemize}
\item Without a server we get the server's rate plus a graph cost that does not grow with $T$ and is set by the farness.
\item We prove the protocol we run (agents act only on published data), for any trigger $\gamma$.
\item The measured cost has the proved shape. Relaying matches consensus with orders of magnitude fewer scalars, and loses only $26\%$ when $90\%$ of link uses fail.
\end{itemize}
\end{block}
\begin{alertblock}{Limitations}
\begin{itemize}
\item Graph lower bound $N\min(T,D)^{1/3}$ vs upper $KF'$: a gap. Bounds linear in $K$; we measure $\approx\sqrt K$.
\item Synthetic data only; private Phase 2 is open. The newest proofs are still being checked.
\end{itemize}
\end{alertblock}
\begin{block}{Future work}
Prove the bin-owner rule (E37); close the graph gap; gap-dependent $K$; private Phase 2; heterogeneous and asynchronous agents.
\end{block}
\end{frame}
