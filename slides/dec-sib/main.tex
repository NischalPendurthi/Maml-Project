\section{Problem}


%====================================================================
\begin{frame}{Problem statement}
\footnotesize
\begin{block}{Setting}
$N$ agents on a graph $\mathcal G$, \textbf{no server}. Each round agent $i$ sees $K$ contexts $x\in\mathbb R^d$, pulls one, observes $y=f(\langle x,\theta^*\rangle)+\eta$. Unknown, shared $\theta^*$ and $f$ ($f$ possibly non-monotone). Goal: small network regret $R_T(N)$, little communication.
\end{block}
\begin{block}{Key difficulties}
\begin{itemize}
\item \textbf{Two unknowns:} the direction needs pooled data; the shape of $f$ needs per-bin counts.
\item \textbf{No coordinator:} totals are built over the graph, $\approx$ diameter rounds late.
\item \textbf{Stale or averaged counts} break the UCB confidence argument.
\item \textbf{Communication:} sharing tables every round costs $\Theta(|E|N_bT)$ scalars.
\end{itemize}
\end{block}
\begin{alertblock}{What prior work misses}
Single-index bandits are single-agent; decentralised bandits assume finite arms, linear or kernel models. No decentralised single-index bandit, and no graph-dependent lower bound for this class.
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Contributions $\star$}
\small
\begin{enumerate}
\setlength{\itemsep}{0.35em}
\item \textbf{New problem:} the first decentralised single-index bandit. Agents share $\theta^*$ and $f$ and talk only over a graph.
\item \textbf{Lower bounds:} no serverless algorithm beats $\Omega((NT)^{2/3})$ (Thm 5). On rings and paths it must also pay $\Omega(N\min(T,D)^{1/3})$ (Thm 7, light cones).
\item \textbf{Upper bounds:} the graph costs only an \emph{additive, horizon-free} term: $\tilde O\big((dNT)^{2/3}+N\Lambda K\big)$ (Thm 22). A provable relay with $\tilde O\big((dNT)^{2/3}+K F'\big)$, where $F'$ is the farness (Thm 27), beats the relay we first ran.
\item \textbf{Communication and robustness:} logarithmic in $T$ for a fixed grid (Thm 29), $O(N_b+N)$ memory per agent. Directed, delayed and time-varying graphs change only the delivery time $\Lambda$, and our bound $B(N-1)$ is tight.
\item \textbf{Experiments:} 16 studies with paired trials and 95\% CIs. Prior papers' algorithms (DLUCB-, ESTOR-, GSTOR- and kernel-style) on our problem. About 12k lines of Python, all figures reproducible.
\end{enumerate}
\end{frame}

%====================================================================
\begin{frame}{Recap: single-index bandits}
\small
\begin{block}{Model}
Each round an agent sees $K$ arms $x_{t,a}\in\mathbb{R}^d$, pulls one, and observes
\[
\boxed{y_t=f(\langle x_t,\theta_*\rangle)+\eta_t}\qquad\theta_*\text{ and }f\text{ unknown},\ f\text{ possibly non-monotone.}
\]
\end{block}
\begin{block}{ZoomSIB-UCB (Dey, Bhore \& Ghosh 2026): optimal $\tilde\Theta(T^{2/3})$}
\begin{itemize}
\item \textbf{Phase 1:} explore; estimate $\theta_*$ with the Stein estimator $\hat v=\frac1{T_0}\sum_t y_tS(x_t)\propto\theta_*$, a \emph{sample mean}.
\item \textbf{Phase 2:} project $z=x^\top\hat\theta$, cut the line into bins, run \emph{sleeping} UCB over the bins available each round. State: $(n_j,S_j)$ per bin.
\end{itemize}
\end{block}
\begin{alertblock}{Our question}
\centering $N$ agents share $\theta_*$ and $f$ but \textbf{no server}: they talk only to graph neighbours.
\end{alertblock}
\end{frame}

\end{block}

\section{Algorithm}


%====================================================================
\begin{frame}{The algorithm: Dec-ZoomSIB (published-only relay)}
\scriptsize
\begin{block}{Every agent $i$ runs the same code}
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{@{}r@{\hspace{0.6em}}l@{}}
\multicolumn{2}{@{}l}{\textbf{Phase 1: learn the direction}}\\
1: & \textbf{for} $t\le T_0$: pull a random arm; $b_i\leftarrow b_i+\varphi_\tau\big(y\,S(x)\big)$ \quad$\triangleright$ truncated Stein sum\\
2: & flood $b_i$; wait $\Lambda$ rounds \quad$\triangleright$ every agent now holds all $N$ sums\\
3: & $\hat\theta\leftarrow\sum_kb_k/\|\sum_kb_k\|_1$ \quad$\triangleright$ identical everywhere; equals the server's\\
\multicolumn{2}{@{}l}{\textbf{Phase 2: learn $f$ with sleeping UCB on bins of width $\Delta$ along $\langle x,\hat\theta\rangle$}}\\
4: & \textbf{for} each round $t$:\\
5: & \quad $(n_j,S_j)\leftarrow\sum_k$ freshest \emph{published} table of $k$ \quad$\triangleright$ own table too, once published\\
6: & \quad pull an arm in the available bin with the largest $S_j/n_j+c_\delta/\sqrt{n_j}$\\
7: & \quad \textbf{if} own unpublished pulls in some bin $j$ exceed $\gamma\max(n_j,1)$: flood own table\\
\end{tabular}
\end{block}
\begin{alertblock}{Why this design}
\begin{itemize}
\item Phase 1 is \textbf{exact}: Stein sums are sufficient statistics, so flooding them gives the server's $\hat\theta$.
\item Phase 2 relays \textbf{exact, sparse} tables, published only when a bin grows by a factor $\gamma$.
\item Acting only on \emph{published} data is the key to the proof (Thm 27).
\end{itemize}
\end{alertblock}
\end{frame}

\section{Theory}


%====================================================================
\begin{frame}{Theory: setting of the proofs}
\footnotesize
\begin{block}{Assumptions (DBG's, with one made explicit)}
\begin{itemize}
\item Contexts i.i.d., coordinates $\kappa$-sub-Gaussian, $\mathbb E[S_j(X)^2]\le M$; $|f|\le L_f$, $|f'|\le L_{f'}$ on $|z|\le X_{\max}$; $\mu^*=\mathbb E f'(\langle X,\theta^*\rangle)>0$.
\item Noise i.i.d.\ and $\sigma$-sub-Gaussian. DBG's Stein lemma already needs i.i.d.\ samples, and their hard instance has $\mathcal N(0,1)$ noise.
\end{itemize}
\end{block}
\begin{block}{Communication: only one number about the graph}
\textbf{Delivery time $\Lambda$:} a flooded message reaches every agent within $\Lambda$ rounds.\\
Static connected graph: $\Lambda=D$ (diameter). Everything below uses the graph \emph{only} through $\Lambda$.
\end{block}
\begin{block}{Results (\texttt{docs/proofs/serverless.pdf})}
Thm 5, 7 lower bounds \;$\cdot$\; Thm 17, 22, 27 regret \;$\cdot$\; Thm 29 communication \;$\cdot$\; Lemma 31 robustness \;$\cdot$\; Thm 33, Prop 34 agreement \& stopping \;$\cdot$\; Thm 35 privacy
\end{block}
\end{frame}

%====================================================================
\begin{frame}{Theorem 5: no serverless algorithm beats $(NT)^{2/3}$}
\footnotesize
\begin{block}{Theorem}
For any $N$-agent algorithm, on any graph, even one where every agent sees all data after every round:\quad
$\displaystyle\sup_{\text{instances}}\mathbb E\,R_T(N)\ \ge\ c\,(NT)^{2/3}.$
\end{block}
\textbf{Proof idea: one agent can simulate the whole network.}
\begin{center}
\begin{tikzpicture}[x=0.5cm,y=0.5cm,font=\scriptsize]
\foreach \t/\r in {0/1,1/2,2/3}{
  \foreach \i in {1,...,4}{
    \node[draw,circle,minimum size=0.45cm,inner sep=0pt,fill=blue!10] at ({\t*5+\i},0) {\i};
  }
  \node at ({\t*5+2.5},1) {round \r};
  \draw[->,thick,orange] ({\t*5+4.4},-0.4) -- ({\t*5+5.6},-0.4) node[midway,below] {rewards};
}
\node[anchor=west] at (15.3,0) {$\cdots$};
\end{tikzpicture}
\end{center}
\begin{itemize}
\item Single-agent round $u=(t-1)N+i$ plays what agent $i$ would play in round $t$.
\item It \textbf{withholds} round $t$'s rewards until all $N$ agents of round $t$ have acted, then runs one communication step.
\item Same law of the history $\Rightarrow$ network regret $=$ regret of one agent over $NT$ rounds $\ge c(NT)^{2/3}$ (DBG Thm 5.1).
\end{itemize}
\vspace{-0.3em}{\tiny Hard instances: DBG's at horizon $NT$ ($d=1$, Gaussian contexts, $K\approx(NT)^{1/3}\log NT$ arms).}
\end{frame}

\begin{frame}{Theorem 7: the graph does cost regret (light cones)}
\scriptsize
\begin{columns}[c]
\begin{column}{0.46\textwidth}
\centering
\begin{tikzpicture}[x=0.24cm,y=0.36cm,font=\tiny]
\draw[->] (0,0) -- (19,0) node[right] {agents};
\draw[->] (0,0) -- (0,5.6) node[above] {round};
\foreach \c in {3,9.5,16}{
  \fill[blue!20] (\c-3,0.2) -- (\c+3,0.2) -- (\c,5) -- cycle;
  \draw[blue!60!black] (\c-3,0.2) -- (\c,5) -- (\c+3,0.2);
  \fill (\c,0.2) circle (1.5pt);
}
\node[align=center] at (9.5,-1.2) {disjoint cones of radius $R$:\\ $\approx R^2$ agent-rounds each};
\end{tikzpicture}
\end{column}
\begin{column}{0.52\textwidth}
\begin{block}{Lemma 6 (light cones are closed)}
Information moves one hop per round, so the actions inside a cone depend only on data inside it.
\end{block}
\end{column}
\end{columns}
\begin{block}{Theorem 7 and Corollary 8}
Each cone acts like one agent with $n\approx R^2$ pulls, so it pays $\Omega(n^{2/3})$. One common hard instance makes all the cones pay at once. On a ring or a path:
\[
\sup\mathbb E\,R_T(N)\ \ge\ c\,\max\Big\{(NT)^{2/3},\ \ N\min(T,D)^{1/3}\Big\}.
\]
\end{block}
\begin{itemize}
\item The second term wins when $T\lesssim\sqrt{ND}$: there, \textbf{no network on a ring can match one agent with $NT$ pulls}.
\item Longer horizons: upper graph term $KF'$ (Thm 27) vs this $ND^{1/3}$. Linear in distance necessary? \alert{Open}: a cone already learns at rate $n^{2/3}$ on its own data.
\end{itemize}
\end{frame}

%====================================================================
\begin{frame}{Theorem 17: regret with synchronised epochs}
\scriptsize
\emph{Synchronised epochs:} a Phase 2 variant in which all agents act on the \emph{same} delayed pooled table plus their own pulls (see backup). It is the first version we could analyse.\vspace{-0.3em}
\begin{block}{Theorem (with probability $\ge1-3\delta$)}
\vspace{-0.8em}
\begin{align*}
R_T(N)\le{}&\underbrace{2L_fN(T_0{+}\Lambda)}_{\text{Phase 1}}
+\underbrace{4\sqrt2\,c_\delta\sqrt{(1{+}N\gamma)N_bNT}}_{\text{UCB}}
+\underbrace{2L_{f'}\Delta NT}_{\text{bin width}}\\[-0.3em]
&+\underbrace{4L_{f'}X_{\max}\varepsilon_0 NT}_{\text{direction error}}
+\underbrace{2L_fN_b\big(2{+}N(\gamma{+}4\Lambda)\big)}_{\text{delay: the only graph term}}
\end{align*}
\end{block}
\begin{block}{Corollary (rate): $\Delta=(NT)^{-1/3}$, $\gamma=\gamma_0/N$, $T_0$ balanced}
\centering $R_T(N)=\tilde O\Big((d\,NT)^{2/3}\;+\;N\Lambda\,(NT)^{1/3}\Big)$\\[0.2em]
\raggedright
Matches Theorem 5 in $N$ and $T$ when $N\Lambda\le(NT)^{1/3}$: \textbf{the graph costs only an additive term.}
\end{block}
\vspace{-0.3em}\textbf{Proof in three steps}
\scriptsize
\begin{enumerate}\setlength{\itemsep}{0pt}
\item \emph{Confidence:} view $=$ common prefix $+$ own block $\Rightarrow$ server width, no $\sqrt N$.
\item \emph{Per pull:} regret $\le 2\times$width $+$ bin width (works for any sleeping pattern).
\item \emph{Stale counts:} $\le\gamma G+4\Lambda$ unseen pulls per agent per bin.
\end{enumerate}
\end{frame}

%====================================================================
\begin{frame}{Theorem 22: the graph's cost does not grow with $T$}
\scriptsize
\begin{block}{Observation}
Theorem 17 lets \emph{every} bin have $\approx4N\Lambda$ unseen pulls (a factor $N_b\sim(NT)^{1/3}$). But a pull into bin $j$ needs an arm in bin $j$, and there are only $K$ arms per agent-round, so in total about $N\Lambda K$ pulls are unseen.
\end{block}
\begin{block}{Theorem 22 (w.p.\ $\ge1-4\delta$; Lemma 21: Chernoff on how often each bin is available)}
\vspace{-0.5em}
\[
\text{delay}\le\underbrace{(2L_f{+}4\sqrt2c_\delta)16N\Lambda\min(K,N_b)}_{\text{graph; no }T,\ N_b}+\underbrace{\tilde O(N_b)}_{\text{no graph}}
\ \Rightarrow\ R_T(N)=\tilde O\big((dNT)^{2/3}+N\Lambda K\big)
\]
\end{block}
\begin{columns}[c]
\begin{column}{0.5\textwidth}
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{@{}lccc@{}}
\toprule
epochs & $NDK$ & measured & ratio\\
\midrule
complete & 320 & 353 & 1.10\\
expander & 960 & 1,030 & 1.07\\
torus & 1,280 & 1,319 & 1.03\\
ring & 2,560 & 2,384 & 0.93\\
path & 4,800 & 4,107 & 0.86\\
\bottomrule
\end{tabular}
\end{column}
\begin{column}{0.48\textwidth}
\begin{itemize}
\item Shape $NDK$ matches the measured excess within $\pm14\%$ on every graph (no fitted constant; the proved constant is larger).
\item Almost flat in $T$, as predicted.
\item Theorem 19 ($\Lambda\to\sqrt\Lambda$) is better when $K$ is large.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

%====================================================================
\begin{frame}{Theorem 27: a provable relay, as good as the run one}
\scriptsize
\begin{block}{The one-line change: every agent acts only on \emph{published} records, including its own}
Then a reward cannot change \emph{when} its pull is published (Lemma 24), and anything it influences reaches every agent later (triangle inequality). So each view is \textbf{one prefix of a fixed order}, a martingale (Lemma 25): no $\sqrt N$, no epochs.
\end{block}
\begin{block}{Theorem 27 ($N\gamma\le1/2$, w.p.\ $\ge1-4\delta$)}
\vspace{-0.6em}
\[
R_T(N)=\tilde O\Big((d\,NT)^{2/3}+K\,F'\Big),\qquad F'=\max_i\sum_l\max(d(i,l),1)\ \ (\text{farness})\le ND
\]
The graph enters through the \textbf{sum of distances}, not $4N\Lambda$; horizon-free. Star $F'\approx2N$, ring $F'\approx ND/2$.
\end{block}
\vspace{-0.3em}
\begin{alertblock}{Measured: the provable relay is better than the one we ran}
Excess over the server at the proved $\gamma=1/(2N)$: complete $-25$ (vs $116$), ring $381$ (vs $528$), path $488$ (vs $633$), for $7$--$8\times$ more scalars. At $\gamma=0.5$ it equals the run relay.
\end{alertblock}
\end{frame}

\begin{frame}{Theorem 29: communication}
\footnotesize
\begin{block}{Theorem}
Under flooding each message crosses each edge at most once. So the number of scalars is
\[
\underbrace{2|E|N(d{+}1)}_{\text{Phase 1}}+\underbrace{2|E|\,|\mathcal E|\,\big(1+N(3N_b{+}1)\big)}_{\text{Phase 2}},
\qquad |\mathcal E|\le1+N_b\big(1+\log_{1+\gamma}(NT)\big).
\]
\end{block}
\begin{itemize}
\item \textbf{Number of epochs:} between two alarms for a bin, its count grows by a factor $>1+\gamma$ $\Rightarrow$ logarithmically many epochs.
\item For a fixed grid this is \textbf{logarithmic in $T$}, against $\Theta(|E|N_bT)$ for per-round sharing.
\end{itemize}
\begin{alertblock}{An honest trade-off}
At the rate-optimal grid, $N_b\sim(NT)^{1/3}$, and with $\gamma=\gamma_0/N$ this becomes $\tilde O(|E|N^2(NT)^{2/3})$: polynomial in $T$.
A constant $\gamma$ needs $N\times$ fewer epochs, but the bound pays $\sqrt{1+N\gamma}$. In experiments $\gamma\le0.5$ costs nothing; only $\gamma=2$ hurts.
\end{alertblock}
\end{frame}

\begin{frame}{Robustness: only $\Lambda$ changes (Lemma 31, Cor 32)}
\small
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\renewcommand{\arraystretch}{1.2}
\footnotesize
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.1cm}l@{}}
\toprule
\textbf{Network} & \textbf{Delivery time $\Lambda$}\\
\midrule
static, connected & $D$\\
directed, strongly connected & directed $D$\\
link delay $\le\tau$ & $\tau D$\\
union of every $B$ rounds connected & $B(N-1)$, \textbf{tight}\\
subgraph stable for $B$ rounds & $D_B$ (its diameter)\\
links fail w.p.\ $q$ & $\approx(D+\log NT)/(1-q)$\\
\bottomrule
\end{tabular}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{Why it is enough}
The proofs use the graph only through $\Lambda$. Records arrive \emph{late}, never \emph{wrong}: the $(NT)^{2/3}$ term never changes, only the additive $N\Lambda(NT)^{1/3}$.
\end{block}
\end{column}
\end{columns}
\vspace{0.6em}
\begin{alertblock}{The diameter of the union graphs does not help}
Each window, the only edges form a star centred at a new agent, all at the window's last round. Every union graph has diameter 2, yet only one new agent is informed per window $\Rightarrow\Lambda=B(N-1)$.
\end{alertblock}
\end{frame}

\begin{frame}{Theorem 33: agents must agree on the direction}
\footnotesize
\begin{block}{Theorem}
If agent $k$ bins with its own $\hat\theta_k$ and $\max_k\|\hat\theta_k-\hat\theta\|_1\le\eta_{\rm agree}$ ($\hat\theta$ = exact pooled), the regret grows by at most
\[
4L_{f'}X_{\max}\,\eta_{\rm agree}\,NT .
\]
\end{block}
\begin{itemize}
\item \textbf{Why:} directions enter the proof only through the projection error. Pooling a bin across agents still pools pulls whose means lie in the same interval.
\item \textbf{Linear in $NT$:} even small disagreement hurts. It matters once $X_{\max}\eta_{\rm agree}NT\sim(NT)^{2/3}$.
\item \textbf{Fix:} one extra flood of the local Stein sums gives every agent the exact direction ($\eta_{\rm agree}=0$), for $\Lambda$ rounds and $2|E|N(d{+}1)$ scalars.
\end{itemize}
\begin{block}{Measured}
Consensus Phase 1: $\times1.25$ (ring), $\times1.34$ (path). One flooding pass: $\times1.08$, $\times1.09$.
\end{block}
\end{frame}

\section{Conclusion}


%====================================================================
\begin{frame}{Conclusion and future work}
\footnotesize
\begin{block}{Conclusion}
\begin{itemize}
\item Without a server, agents learn an unknown index and link at the server's rate, plus an \textbf{additive graph cost that does not grow with $T$}.
\item That cost is set by distances (farness); the measured cost has the proved shape.
\item Relaying exact, sparse statistics: the regret of consensus, orders of magnitude fewer scalars, robust to failing links.
\end{itemize}
\end{block}
\begin{alertblock}{Limitations}
\begin{itemize}
\item Graph lower bound $N\min(T,D)^{1/3}$ vs upper $KF'$: a gap. Bounds linear in $K$; we measure $\approx\sqrt K$.
\item Synthetic data only; private Phase 1 is costly; private Phase 2 is open. The newest proofs are still being checked.
\end{itemize}
\end{alertblock}
\begin{block}{Future work}
Close the graph gap; gap-dependent $K$; private Phase 2; heterogeneous ($\theta^*_i$) and asynchronous agents.
\end{block}
\end{frame}
