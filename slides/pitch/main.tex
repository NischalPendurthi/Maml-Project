\section{Motivation}
%=====================================================================
\begin{frame}{The assumption everybody makes, and why it breaks}
\begin{frame}{Contextual Bandits with Unknown Links}

  Contextual bandits assume we know how context maps to reward:
  Standard contextual bandits assume a known relationship between context and reward:
  \[
    \mathbb{E}[y \mid x] \;=\; \underbrace{\langle x, \theta_*\rangle}_{\text{linear bandit}}
    \qquad\text{or}\qquad
    \underbrace{f(\langle x, \theta_*\rangle)}_{\substack{\text{generalized linear}\\ f \text{ \textbf{known}}}}
    \underbrace{f(\langle x, \theta_*\rangle)}_{\substack{\text{generalized linear}\\ f \text{ is known}}}
  \]

  \pause
  \begin{block}{But the link function is essentially never known}
    Dose--response, congestion, click-through and user-satisfaction curves are
    non-monotone and unknown in advance.
  \begin{block}{Unknown and non-monotone reward profiles}
    In practical applications like pricing, traffic routing, and dose-response,
    the mapping $f$ is rarely known beforehand and often non-monotone.
  \end{block}

  \pause
  \begin{alertblock}{This is not a mild modelling error}
    Misspecifying $f$ makes state-of-the-art algorithms incur \textbf{linear regret} ---
    they do not degrade gracefully, they fail.\footnotemark
  \begin{alertblock}{Consequence of misspecification}
    When $f$ is misspecified, standard methods like LinUCB or GLM-UCB suffer
    \textbf{linear regret} $\Omega(T)$ rather than sublinear growth.\footnotemark
  \end{alertblock}
  \footnotetext[1]{Ghosh, Chowdhury \& Gopalan, \emph{Misspecified Linear Bandits}, AAAI 2017.}

\end{frame}

\section{Problem Statement}
%=====================================================================
\begin{frame}{Problem statement: the single-index bandit (SIB)}
\begin{frame}{Single-Index Bandits (SIB)}
  \small

  \begin{block}{Model}
    At round $t$ the learner sees $K$ arms $\mathcal{X}_t=\{x_{t,a}\}_{a\in[K]}$ drawn
    i.i.d.\ from a density $p$ on $\mathbb{R}^d$; it pulls $x_t$ and observes
  \begin{block}{System model}
    At round $t$, the learner observes $K$ arms $\mathcal{X}_t=\{x_{t,a}\}_{a\in[K]}$ drawn
    i.i.d.\ from density $p$ on $\mathbb{R}^d$, chooses $x_t$, and receives reward
    \[
      y_t \;=\; f(\langle x_t,\theta_*\rangle) + \eta_t,
      \qquad \eta_t \ \sigma\text{-sub-Gaussian},
    \]
    with \textbf{both} $\theta_*\in\mathbb{R}^d$ \textbf{and} $f:\mathbb{R}\to\mathbb{R}$
    \textbf{unknown}. Normalise $\|\theta_*\|_1=1$.
    where both the index direction $\theta_*\in\mathbb{R}^d$ and the link function $f:\mathbb{R}\to\mathbb{R}$
    are unknown. We normalize $\|\theta_*\|_1=1$.
  \end{block}

  \pause
  \alert{$f$ need not be monotone} $\Rightarrow$ the best arm is \emph{not} the one with
  the largest projection.
  \alert{Since $f$ is arbitrary and non-monotone}, picking the arm with the largest projection $\langle x, \theta_*\rangle$ is generally suboptimal.

  \pause
  \begin{block}{Key tool --- Stein's identity, with score $S(x)=-\nabla\log p(x)$}
  \begin{block}{Estimating $\theta_*$ via Stein's identity with score $S(x)=-\nabla\log p(x)$}
    \[
      \mathbb{E}\big[\,y\,S(x)\,\big]
      \;=\; \mathbb{E}\big[f'(\langle x,\theta_*\rangle)\big]\,\theta_*
      \;=\; \mu_*\,\theta_*
    \]
    The direction is recoverable by a \textbf{plain sample average} --- with no knowledge
    of $f$ whatsoever.
    The expectation isolates the direction $\theta_*$ up to a scalar constant $\mu_*$.
    This lets us estimate $\theta_*$ directly using empirical sample averages without fitting $f$.
  \end{block}

\end{frame}

\section{Our Problem}
%=====================================================================
\begin{frame}{Our problem: what does collaboration buy?}
\begin{frame}{Problem Formulation: Multi-Agent SIB}

  \begin{block}{Setting}
    $N$ agents on a connected graph $G$ with communication matrix $\mathbf{P}$.
    Each round, every agent $i$ sees its own arm set, pulls $x_{i,t}$, observes
    Consider $N$ decentralized agents on a connected communication graph $G$.
    At each round $t$, each agent $i$ observes an independent arm set, pulls $x_{i,t}$, and observes
    $y_{i,t}=f(\langle x_{i,t},\theta_*\rangle)+\eta_{i,t}$.
    $f$ and $\theta_*$ are \textbf{shared and unknown}.
    Agents exchange $\le B$ values with neighbours every $C$ rounds.
    The parameter $\theta_*$ and link $f$ are identical across agents but unknown.
    Agents can exchange messages with their neighbors subject to a communication budget.
  \end{block}

  \begin{block}{Objective --- network regret}
  \begin{block}{Objective: Cumulative Network Regret}
    \vspace{-0.6em}
    \[
      R_T(N)\;=\;\sum_{t=1}^{T}\sum_{i=1}^{N}
      \Big[ f(\langle x^*_{i,t},\theta_*\rangle) - f(\langle x_{i,t},\theta_*\rangle) \Big]
    \]
  \end{block}

  \pause
  \begin{center}
    \large
    \emph{How much does cooperation help when the reward function itself\\
    must be learned --- and at what communication cost?}
    \emph{Can decentralized collaboration improve per-agent regret when neither the index direction nor the link function is known?}
  \end{center}

\end{frame}

\section{Approach}
%=====================================================================
\begin{frame}{Why this algorithm invites federation}
\begin{frame}{Two Sharing Schemes in ZoomSIB-UCB}

  \small
  \textsc{ZoomSIB-UCB} (Dey, Bhore \& Ghosh '26) is two-phase, and each phase is
  shareable --- but by a \emph{different mechanism}.
  \textsc{ZoomSIB-UCB} (Dey, Bhore \& Ghosh '26) operates in two sequential phases.
  Each phase provides a distinct point of collaboration:

  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Phase 1 --- estimate $\theta_*$}
      \begin{block}{Phase 1: Estimating $\theta_*$}
        \footnotesize
        Truncated Stein estimator over $T_0$ rounds.\\[0.3em]
        \alert{It is a \textbf{sample mean}} --- so averaging local estimators is
        \textbf{algebraically identical} to the centralized estimator on pooled
        data.\\[0.3em]
        Exact federation, \textbf{one} message of $d$ numbers, no consensus
        iterations, no approximation.
        Each agent runs a truncated Stein estimator over $T_0$ initial rounds.\\[0.3em]
        Because the estimator is a sample mean, averaging local estimates across agents produces the exact pooled centralized estimate in a single message ($d$ floats), with no iterative consensus.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Phase 2 --- UCB over bins}
      \begin{block}{Phase 2: Binned UCB}
        \footnotesize
        $N_{\mathrm{bins}}=O(T^{1/3})$ bins on the index line; \emph{sleeping} UCB.\\[0.3em]
        The entire learner state is \textbf{two scalars per bin} $(n_j,S_j)$ ---
        so sharing it is a standard cooperative MAB over the graph $G$.\\[0.3em]
        Cheap to communicate, but the pooling is statistical, not exact.
        The projected line $[-1, 1]$ is split into $N_{\mathrm{bins}}=O(T^{1/3})$ intervals, running a sleeping UCB.\\[0.3em]
        Each agent maintains two running scalars per bin $(n_j,S_j)$. Sharing these statistics across graph $G$ corresponds to a cooperative multi-armed bandit with statistical pooling.
      \end{block}
    \end{column}
  \end{columns}

  \pause
  \vspace{0.2em}
  \vspace{0.4em}
  \footnotesize
  Prop.~4.7 of Dey et al.\ is a \textbf{modular reduction}: \emph{any} sleeping-bandit
  subroutine may be substituted into Phase 2 --- and a cooperative one is exactly that.
  Proposition 4.7 in Dey et al.\ proves a modular reduction: any valid sleeping-bandit subroutine can be used in Phase 2 while preserving regret guarantees. We exploit this by plugging in a cooperative sleeping bandit.

\end{frame}

\begin{frame}{What we set out to answer}
\begin{frame}{Core Research Questions}

  We are \emph{not} claiming a rate in advance. The algorithm splits into two
  independently shareable pieces; how much each one buys is exactly what we
  intend to measure.
  Rather than assuming theoretical gains upfront, we examine how the two phases contribute to speedup under bandwidth limits.

  \vspace{0.4em}
  \begin{block}{Research questions}
  \begin{block}{Questions under investigation}
    \small
    \begin{enumerate}
      \item Does sharing reduce \textbf{per-agent} regret in a single-index bandit,
            and by how much?
      \item The two phases pool very differently --- an \emph{exact} average versus a
            statistical one. \alert{Does the benefit split unevenly between them?}
      \item \textbf{How little communication is enough?} Phase 1 needs a single
            message; can Phase 2 be made nearly as cheap?
      \item How does any gain scale with $N$, with dimension $d$, and with the
            graph topology (spectral gap $1-|\lambda_2|$)?
      \item \textbf{Per-agent regret scaling:} Does collaboration yield the expected $O(1/\sqrt{N})$ per-agent regret reduction?
      \item \textbf{Phase asymmetry:} Phase 1 pooling is algebraically exact, while Phase 2 pooling is statistical. Which phase drives the empirical performance gain?
      \item \textbf{Communication budgets:} How far can message frequency be reduced (via periodic, event-triggered, or quantized exchanges) before regret degrades?
      \item \textbf{Network topology:} How sensitive is performance to the graph spectral gap $1-|\lambda_2(\mathbf{P})|$ across different dimensions $d$?
    \end{enumerate}
  \end{block}

  \pause
  \footnotesize
  \textbf{If time permits.} \textbf{(a)} \emph{Safety}: an unknown constraint
  $h(\langle x,\mu_*\rangle)\le c$.
  \textbf{(b)} \emph{Clustered indices}: agents in $M$ unknown groups, IFCA-style.
  \textbf{Potential extensions:} (a) Safety constraints with unknown threshold functions $h(\langle x,\mu_*\rangle)\le c$; (b) clustered index directions where agents belong to unknown latent groups.

\end{frame}
