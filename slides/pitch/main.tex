\section{Motivation}
%====================================================================
\begin{frame}{From contextual bandits to an unknown reward function}
\small
A contextual bandit chooses an action $x$ using its context and observes a noisy reward $y$.
\begin{block}{Common models}
\[
\mathbb{E}[y\mid x]=\langle x,\theta_*\rangle\quad\text{(Linear)},\qquad \mathbb{E}[y\mid x]=f(\langle x,\theta_*\rangle)\quad\text{(GLM)}.
\]
\end{block}
\begin{center}
\[
x\in\mathbb{R}^d\quad\longrightarrow\quad x^\top\theta_*\quad\longrightarrow\quad f(\cdot)\quad\longrightarrow\quad y
\]
\end{center}
\begin{alertblock}{The gap}
In GLMs, $f$ is known. In many applications, the reward curve is unknown and may be non-monotone.
\[
\boxed{\text{What if both }\theta_*\text{ and }f\text{ are unknown?}}
\]
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Single-Index Bandits: the problem}
\small
\begin{block}{Model}
At round $t$, the learner observes $K$ arms $\mathcal{X}_t=\{x_{t,a}\}_{a\in[K]}$, chooses $x_t$, and observes
\[
\boxed{y_t=f(\langle x_t,\theta_*\rangle)+\eta_t}
\]
where $\theta_*\in\mathbb{R}^d$ and $f:\mathbb{R}\rightarrow\mathbb{R}$ are both unknown, with $\|\theta_*\|_1=1$.
\end{block}
\begin{center}
\[
x_t\in\mathbb{R}^d\quad\xrightarrow{\ \theta_*\ }\quad x_t^\top\theta_*\quad\xrightarrow{\ f\ }\quad y_t
\]
\end{center}
\begin{alertblock}{Why is this hard?}
If $f$ is non-monotone,
\[
x_1^\top\theta_*>x_2^\top\theta_*\quad\not\Rightarrow\quad f(x_1^\top\theta_*)>f(x_2^\top\theta_*).
\]
\textbf{The largest projection need not be the best arm.}
\end{alertblock}
\end{frame}

%====================================================================
\begin{frame}{Key idea: estimate $\theta_*$ without knowing $f$}
\small
Define the score function
\[
S(x)=-\nabla\log p(x).
\]
Under Stein's identity,
\[
\mathbb{E}[yS(x)]=\mathbb{E}\!\left[f'(\langle x,\theta_*\rangle)\right]\theta_*.
\]
Let
\[
\mu_*=\mathbb{E}\!\left[f'(\langle x,\theta_*\rangle)\right].
\]
Then
\[
\boxed{\mathbb{E}[yS(x)]=\mu_*\theta_*}
\]
so the direction of
\[
\boxed{\hat v=\frac{1}{T_0}\sum_{t=1}^{T_0}y_tS(x_t)}
\]
estimates the direction of $\theta_*$ without estimating $f$.
\begin{block}{Key insight}
\centering
\textbf{Unknown $f$ does not prevent us from estimating the index direction.}
\end{block}
\end{frame}

%====================================================================
\begin{frame}{The SIB problem reduces to 1-D learning}
\small
Once $\theta_*$ is estimated, project each context onto the estimated direction:
\[
z=x^\top\hat\theta.
\]
\begin{center}
\[
x\in\mathbb{R}^d\quad\xrightarrow{\ \hat\theta\ }\quad z=x^\top\hat\theta\in\mathbb{R}\quad\xrightarrow{\ f\ }\quad f(z)
\]
\end{center}
The original $d$-dimensional problem becomes learning an unknown function on a 1-D index.
\[
[-1,1]=I_1\cup I_2\cup\cdots\cup I_M,\qquad M=O(T^{1/3}).
\]
\begin{block}{ZoomSIB-UCB}
The index domain is divided into bins, and a sleeping-UCB procedure learns which regions of the unknown reward curve are promising.
\end{block}
\end{frame}

%====================================================================
\section{Federation}
%====================================================================
\begin{frame}{Problem Formulation: Multi-Agent SIB}
\small
\begin{block}{Setting}
Consider $N$ decentralized agents on a connected communication graph $G$. At each round $t$, each agent $i$ observes an independent arm set, pulls $x_{i,t}$, and observes $y_{i,t}=f(\langle x_{i,t},\theta_*\rangle)+\eta_{i,t}$. The parameter $\theta_*$ and link $f$ are identical across agents but unknown. Agents can exchange messages with their neighbors subject to a communication budget.
\end{block}
\begin{block}{Objective: Cumulative Network Regret}
\[
R_T(N)=\sum_{t=1}^{T}\sum_{i=1}^{N}\left[f(\langle x^*_{i,t},\theta_*\rangle)-f(\langle x_{i,t},\theta_*\rangle)\right].
\]
\end{block}
\begin{center}
\emph{Can decentralized collaboration improve per-agent regret when neither the index direction nor the link function is known?}
\end{center}
\end{frame}

%====================================================================
\begin{frame}{Fed-ZoomSIB: federate both phases}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Phase 1: Estimate $\theta_*$}
Each agent computes
\[
\hat\theta_i\propto\frac{1}{T_0}\sum_t y_{i,t}S(x_{i,t}).
\]
Then agents exchange their local estimates:
\[
\boxed{\hat\theta=\frac{1}{N}\sum_{i=1}^{N}\hat\theta_i}
\]
\textbf{Exact pooled averaging}\\[-0.2em]
One $d$-dimensional message.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Phase 2: Learn $f$}
Project:
\[
z_{i,t}=x_{i,t}^{\top}\hat\theta.
\]
Each agent maintains bin statistics
\[
(n_j,S_j).
\]
Agents periodically exchange these statistics.
\textbf{Statistical pooling.}
\end{block}
\end{column}
\end{columns}
\vspace{0.2em}
\begin{center}
\boxed{\text{Phase 1: exact pooling}\qquad\text{vs.}\qquad\text{Phase 2: statistical pooling}}
\end{center}
\end{frame}

%====================================================================
\begin{frame}{What do we want to find out?}
\small
\begin{block}{1. Does collaboration help?}
Compare per-agent regret under independent and federated learning:
\[
R_T^{\mathrm{federated}}\quad\text{vs.}\quad R_T^{\mathrm{independent}}.
\]
\end{block}
\begin{block}{2. How much communication is enough?}
Study the tradeoff
\[
\boxed{\text{Regret}\quad\leftrightarrow\quad\text{Communication}}
\]
using periodic, event-triggered, and quantized exchanges.
\end{block}
\begin{block}{3. How does the network matter?}
Study dependence on the number of agents, dimension, number of arms, and graph connectivity:
\[
N,\qquad d,\qquad K,\qquad 1-|\lambda_2(\mathbf P)|.
\]
\end{block}
\end{frame}
